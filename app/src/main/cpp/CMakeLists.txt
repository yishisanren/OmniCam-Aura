cmake_minimum_required(VERSION 3.22.1)
project(aura_native CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(aura_native SHARED native_entry.cpp gr_basictone.cpp pop_path_remap.cpp)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../assets/device_profile.json" AURA_PROFILE)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../assets/device_profile.json")
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/../assets/device_profile.json" AURA_PROFILE_SHA256)
string(JSON AURA_MODEL GET "${AURA_PROFILE}" model)
string(JSON AURA_FIRMWARE GET "${AURA_PROFILE}" firmware)
target_compile_definitions(aura_native PRIVATE AURA_MODEL="${AURA_MODEL}" AURA_FIRMWARE="${AURA_FIRMWARE}" AURA_PROFILE_SHA256="${AURA_PROFILE_SHA256}")
target_compile_options(aura_native PRIVATE -O2 -fvisibility=hidden)
target_link_libraries(aura_native log dl)
